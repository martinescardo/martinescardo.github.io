K-∀-shift : {R : Ω}
            {A : ℕ → Ω}
          → (∀(n : ℕ) → R → A n)                               -- efqs,
          → (∀(n : ℕ) → K {R} (A n)) → K {R} (∀(n : ℕ) → A n)  -- shift.
K-∀-shift = g choice
 where
  g : ℕ → _
  g O               = K-∀-shift-selection
  g 1               = K-∀-shift-mbr
  g (succ (succ _)) = K-∀-shift-bbc
